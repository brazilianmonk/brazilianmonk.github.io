## Exercise

1.  instrumental case 
    -   tisaraṇena saddhiṃ pañcasīlaṃ dhammaṃ – **sahādiyogattha**
    -   sugandhinā'haṃ gandhena pūjayāmi tathāgataṃ –  **karaṇattha**
    -   pūjemi buddhaṃ kusumena'nena – **karaṇattha**
    -   puññena me'tena labhāmi mokkhaṃ – **karaṇattha**
    -   amhehi sambhataṃ puññasampadaṃ – **kattvattha**
2.  dative case 
    -   namo tassa bhagavato arahato – **sampadānattha**
    -   sabbe devā anumodantu, sabbasampattisiddhiyā – **sampadānattha**
    -   senāsanaṃ paṭisevāmi sītassa paṭighātāya, uṇhassa paṭighātāya  &#x2013; **tadatthasampadāna**
3.  ablative case 
    -   adinnādānā veramaṇī-sikkhāpadaṃ – **apādānattha (virati)**
    -   avijjāpaccayā saṅkhārā, saṅkhārapaccayā viññāṇaṃ – \*hetvattha \*
    -   āsavakkhayā'vahaṃ hotu, sabbadukkhā pamuñcatu – **apādānattha**
    -   sataṃ samāgamo hotu, yāva nibbānappattiyā – **apādānattha (with yāva)**
4.  genitive / locative case 
    -   supaṭipanno bhagavato sāvakasaṅgho – **sambandhattha chaṭthī**
    -   yo vadataṃ pavaro manujesu – **niddhāraṇattha chaṭṭhī**
    -   sabbe bhāyanti maccuno – **apādānattha chaṭṭhī**
    -   yattha ca dinna(ṁ) mahapphalamāhu, catusu sucīsu purisayugesu &#x2013; **sampadānattha sattamī**
    -   imesaṃ tesattatiyā ñāṇānaṃ sattasaṭṭhiñāṇāni sāvakāsādhāraṇāni – **niddhāraṇattha chaṭṭhī**
    -   imameva kāyaṃ pūraṃ nānappakārassa asucino – **karaṇattha chaṭṭhī**
    -   ācariyo me bhante hohi, āyasmato nissāya vacchāmi &#x2013; **kammattha chaṭṭhī**
    -   imasmiṃ sati idaṃ hoti - **bhavalakkhaṇa sattamī**
    -   surāmerayapāne pācittiyaṃ – **nimittattha sattamī**

        .replace(/[^\w\s-]/g, "")
        .replace(/\s+/g, "-");
    }
    
    if (heading.tagName === "H2") {
      // Create container for this H2 section
      const li = document.createElement("li");
      li.className = "toc-h2-item";
      
      // Add toggle arrow
      const toggle = document.createElement("span");
      toggle.className = "toc-toggle";
	toggle.textContent = "▶";  // Collapsed by default
      toggle.setAttribute("aria-label", "Collapse section");
      
      // Add the H2 link
      const a = document.createElement("a");
      a.href = `#${heading.id}`;
      a.textContent = heading.textContent;
      a.className = "toc-h2-link";
      
      // Container for H3 items (to be filled later)
      const h3Container = document.createElement("ul");
      h3Container.className = "toc-h3-list";
	h3Container.classList.add("collapsed");
      
	// Assemble
      li.appendChild(toggle);
      li.appendChild(a);
      li.appendChild(h3Container);
      tocList.appendChild(li);
      
      // Store references
      currentH2Item = li;
      currentH3List = h3Container;
      
      // Add click toggle functionality
      const toggleSection = () => {
        const isCollapsed = h3Container.classList.contains("collapsed");
        if (isCollapsed) {
          h3Container.classList.remove("collapsed");
          toggle.textContent = "▼";
          toggle.setAttribute("aria-label", "Collapse section");
        } else {
          h3Container.classList.add("collapsed");
          toggle.textContent = "▶";
          toggle.setAttribute("aria-label", "Expand section");
        }
      };
      
      toggle.addEventListener("click", function(e) {
        e.preventDefault();
        e.stopPropagation();
        toggleSection();
      });
      
      a.addEventListener("click", function(e) {
        // Allow normal anchor behavior, but also toggle if desired?
        // Comment out the next line if you want clicking the link to ALSO toggle
        // e.preventDefault(); 
        // Uncomment below to toggle when clicking the link text too
        // toggleSection();
        // Then scroll to heading
        // document.getElementById(heading.id).scrollIntoView({ behavior: "smooth" });
      });
      
    } else if (heading.tagName === "H3" && currentH3List) {
      // Add H3 item under current H2
      const li = document.createElement("li");
      const a = document.createElement("a");
      a.href = `#${heading.id}`;
      a.textContent = heading.textContent;
      li.appendChild(a);
      currentH3List.appendChild(li);
    }
  });
  
  // Insert TOC at the beginning of the page
  const firstHeading = document.querySelector("h1, h2");
  if (firstHeading) {
    firstHeading.parentNode.insertBefore(toc, firstHeading);
  } else {
    document.body.insertBefore(toc, document.body.firstChild);
  }
});
